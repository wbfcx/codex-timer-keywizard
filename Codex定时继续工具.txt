Function Codex_计算延迟毫秒(选择结果)//0600

    Dim 结果数组, 时间数组//0601
    Dim 目标小时, 目标分钟//0602
    Dim 当前秒数, 目标秒数, 相差秒数//0603

    Codex_计算延迟毫秒 = 0//0604

    结果数组 = Split(选择结果, "|")//0605

    If UBound(结果数组) < 1 Then//0606
        Exit Function//0607
    End If//0608

    时间数组 = Split(结果数组(1), ":")//0609

    If UBound(时间数组) < 1 Then//0610
        Exit Function//0611
    End If//0612

    If IsNumeric(时间数组(0)) = False Then//0613
        Exit Function//0614
    End If//0615

    If IsNumeric(时间数组(1)) = False Then//0616
        Exit Function//0617
    End If//0618

    目标小时 = CInt(时间数组(0))//0619
    目标分钟 = CInt(时间数组(1))//0620

    当前秒数 = Hour(Now()) * 3600 + Minute(Now()) * 60 + Second(Now())//0621
    目标秒数 = 目标小时 * 3600 + 目标分钟 * 60//0622

    相差秒数 = 目标秒数 - 当前秒数//0623

    If 相差秒数 <= 0 Then//0624
        相差秒数 = 相差秒数 + 86400//0625
    End If//0626

    Codex_计算延迟毫秒 = 相差秒数 * 1000//0627

End Function//0628
